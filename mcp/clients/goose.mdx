<Steps>
  <Step title="Add an extension">
    On desktop, open **Settings → Extensions → Add**. On the CLI, run
    `goose configure` and choose **Add Extension**.
  </Step>
  <Step title="Choose the remote type">
    Pick **Remote Extension (Streamable HTTP)** and paste:

    ```
    https://mcp.manypi.com/mcp
    ```
  </Step>
  <Step title="Add your key">
    Add a request header:

    ```
    Authorization: Bearer mpi_your_api_key
    ```
  </Step>
</Steps>
